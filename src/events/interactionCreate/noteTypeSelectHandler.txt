const noteStore = require('../../utils/noteStorage');

module.exports  = async (client, interaction) =>{
    if (!interaction.isStringSelectMenu()) return;
    if (interaction.customId !== 'note-type-select') return;

    const type = interaction.values[0];

    noteStore.updatePending(interaction.user.id, {
        type,
        step: 'writeText'
    });

    await interaction.showModal({
        custom_id: `leave-note-modal`,
        title: 'Write Your Note',
        components: [
            {
                type: 1,
                components: [
                    {
                        type: 4,
                        custom_id: 'note-text',
                        label: 'Your note',
                        style: 2,
                        required: true
                    }
                ]
            }
        ]
    });
};
